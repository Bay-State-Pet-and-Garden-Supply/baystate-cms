### 1. Server Route: `POST /api/onboarding/settings/domain-diagnostics/:domain/generate-profile`

In `src/server/routes/onboarding-routes.ts`, add a new route near the domain-diagnostics GET route:

```typescript
route.post('/onboarding/settings/domain-diagnostics/:domain/generate-profile', async (c) => {
  const domain = c.req.param('domain');
  // 1. Check env flag
  if (!isProfileGenerationEnabled()) {
    return c.json({ error: 'Profile generation is not enabled. Set SHOPSITE_CMS_PROFILE_GENERATION_ENABLED=true.' }, 400);
  }
  // 2. Get up to 3 cached sitemap URLs for this domain
  const allCaches = listAllSitemapCaches();
  const cache = allCaches.find(row => row.domain === domain);
  if (!cache || cache.sitemapUrlsCount === 0) {
    return c.json({ error: 'No cached sitemap URLs found for this domain. Run source discovery first.' }, 400);
  }
  const sampleUrls = cache.urls.slice(0, 3);
  
  // 3. For each URL, fetch HTML and run the generator
  const generationIds: string[] = [];
  for (const url of sampleUrls) {
    try {
      const response = await fetch(url, { headers: HTTP_EXTRACTION_HEADERS, signal: AbortSignal.timeout(15000) });
      if (!response.ok) continue;
      const html = await response.text();
      const generated = await generateExtractorProfile(url, html, { domain, sourceUrl: url });
      if (generated) {
        const rec = insertProfileGeneration({
          domain,
          sourceUrl: url,
          expectedName: null,
          brandHint: null,
          selectors: generated as unknown as Record<string, unknown>,
          status: 'proposed',
          confidence: 0.5,
          llmProvider: null,
          llmModel: null,
          errorMessage: null,
        });
        generationIds.push(rec.id);
      }
    } catch (err) {
      console.warn(`[GenerateProfile] Failed for ${url}:`, err);
    }
  }
  
  return c.json({ 
    success: generationIds.length > 0,
    generationIds,
    sampledUrls: sampleUrls.length,
    domain,
  });
});
```

You'll need to import: `isProfileGenerationEnabled` and `generateExtractorProfile` from `../../onboarding/profile-generator`, `insertProfileGeneration` from `../../db/repositories/profile-generation-repo`, `listAllSitemapCaches` from `../../db/repositories/sitemap-cache-repo`, and `HTTP_EXTRACTION_HEADERS` from `../../onboarding/page-extractor`.

### 2. Client API: `generateProfileForDomain(domain)`

In `src/client/onboarding-api.ts`, add:

```typescript
export interface GenerateProfileResult {
  success: boolean;
  generationIds: string[];
  sampledUrls: number;
  domain: string;
}

export async function generateProfileForDomain(domain: string): Promise<GenerateProfileResult> {
  return request<GenerateProfileResult>(`/settings/domain-diagnostics/${encodeURIComponent(domain)}/generate-profile`, {
    method: 'POST',
  });
}
```

In `src/client/components/OnboardingSettings.tsx`:

```typescript
const handleGenerateProfile = async (domain: string) => {
  setGeneratingProfileDomain(domain);
  setError('');
  try {
    const result = await generateProfileForDomain(domain);
    if (result.generationIds.length > 0) {
      alert(`Generated ${result.generationIds.length} profile proposal(s) for ${domain}. Review them in Generated Profile Governance.`);
      // Scroll to governance section
      document.getElementById('generated-profile-governance')?.scrollIntoView({ behavior: 'smooth' });
      // Refresh diagnostics to update generation counts
      void loadDomainDiagnostics();
    } else {
      setError(`Profile generation for ${domain} returned no proposals. The LLM may not have produced valid selectors from the sampled pages.`);
    }
  } catch (err) {
    setError(err instanceof Error ? err.message : String(err));
  } finally {
    setGeneratingProfileDomain(null);
  }
};
```
